function toggleDD(ddId,fbId,e){
  e.stopPropagation();
  const dd=document.getElementById(ddId), fb=document.getElementById(fbId);
  if(openDD&&openDD!==dd){openDD.style.display='none';if(openFB)openFB.classList.remove('on');}
  const isOpen=dd.style.display==='block';
  dd.style.display=isOpen?'none':'block';
  fb.classList.toggle('on',!isOpen);
  document.getElementById('ov').classList.toggle('on',!isOpen);
  openDD=isOpen?null:dd; openFB=isOpen?null:fb;
}

function closeAll(){
  document.querySelectorAll('.dd').forEach(d=>d.style.display='none');
  document.querySelectorAll('.fb').forEach(b=>b.classList.remove('on'));
  document.getElementById('ov').classList.remove('on');
  openDD=null; openFB=null;
}

function selectRep(val,lblId,ddId,fbId){
  document.getElementById(lblId).textContent=val;
  closeAll();
  showToast('Filter: '+val);
}

function selectChip(el,val,lblId){
  document.querySelectorAll('.dq-chip').forEach(c=>{
    c.style.borderColor='var(--border)';c.style.color='var(--t2)';c.style.background='white';c.style.fontWeight='normal';
  });
  el.style.borderColor='var(--red)';el.style.color='var(--red)';el.style.background='#fff5f5';el.style.fontWeight='600';
  document.getElementById(lblId).textContent=val;
}

function filterReps(q){
  const items=document.querySelectorAll('#rep-list .dd-item');
  items.forEach(item=>{
    item.style.display=item.textContent.toLowerCase().includes(q.toLowerCase())?'flex':'none';
  });
}

function openCard(name){
  showToast('Opening deal: '+name);
}

function showToast(msg){
  const t=document.createElement('div');
  t.className='toast';t.textContent=msg;
  document.body.appendChild(t);
  setTimeout(()=>t.remove(),2400);
}
